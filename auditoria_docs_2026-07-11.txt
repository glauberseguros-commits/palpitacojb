00550:     byHour,

00553:     removedSoftApplied,

00569:

00571:       return { hour: hh, kind, status: "gap", reason: "source_gap", draw: null };
00572:     }
00573:

00575: 
00576:     if (
00577:       capToday &&
00578:       isToday &&
00579:       ((expectedHard && expectedHard.includes(hh)) || (expectedSoft && expectedSoft.includes(hh)))
00580:     ) {
00581:       const published = isSlotPublishedToday(hh, nowMinBR, slotGraceMin);
00582:       if (!published) {
00583:         return { hour: hh, kind, status: "future", reason: "not_yet_published", draw: null };
00584:       }
00585:     }
00586: 
00587:     if (expectedHardPublished && expectedHardPublished.includes(hh)) {
00588:       return { hour: hh, kind: "hard", status: "missing", draw: null };
00589:     }
00590:     if (expectedSoftPublished && expectedSoftPublished.includes(hh)) {
00591:       return { hour: hh, kind: "soft", status: "soft_missing", draw: null };
00592:     }
00593: 
00594:     return { hour: hh, kind, status: "gap", reason: "not_expected", draw: null };
00595:   });
