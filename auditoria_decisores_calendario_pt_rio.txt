00045:     if (!Number.isFinite(hh) || hh < 0 || hh > 23) return null;
00046:     if (!Number.isFinite(mm) || mm < 0 || mm > 59) return null;
00047:     if (!Number.isFinite(ss) || ss < 0 || ss > 59) return null;
00048:     return `${pad2(hh)}:${pad2(mm)}`;
00049:   }
00050: 
00051:   // HH:MM (aceita minuto 1-2 dígitos, ex: 11:9)
00052:   const m0 = s.match(/^(\d{1,2}):(\d{1,2})$/);
00053:   if (m0) {
00054:     const hh = Number(m0[1]);
00055:     const mm = Number(m0[2]);
00056:     if (!Number.isFinite(hh) || hh < 0 || hh > 23) return null;
00057:     if (!Number.isFinite(mm) || mm < 0 || mm > 59) return null;
00058:     return `${pad2(hh)}:${pad2(mm)}`;
00059:   }
00060: 
00061:   // "11h" / "11hs" / "11"
00062:   const m2 = s.match(/^(\d{1,2})(?:\s*h(?:s)?)?$/i);
00063:   if (m2) {
00064:     const hh = Number(m2[1]);
00065:     if (!Number.isFinite(hh) || hh < 0 || hh > 23) return null;
00066:     return `${pad2(hh)}:00`;
00067:   }
00068: 
00069:   // fallback: pega o primeiro número como hora
00070:   const m3 = s.match(/(\d{1,2})/);
00071:   if (m3) {
00072:     const hh = Number(m3[1]);
00073:     if (!Number.isFinite(hh) || hh < 0 || hh > 23) return null;
00074:     return `${pad2(hh)}:00`;
00075:   }
00076: 
00077:   return null;
00078: }
