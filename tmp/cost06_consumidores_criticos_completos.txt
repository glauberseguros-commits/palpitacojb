   383: function dedupeDrawsLocal(draws) {
   384:   const arr = Array.isArray(draws) ? draws : [];
   385: 
   386:   const byKey = new Map();
   387:   const order = [];
   388: 
   389:   function score(d) {
   390:     const prizesLen = Array.isArray(d?.prizes) ? d.prizes.length : 0;
   391:     const pc = Number.isFinite(Number(d?.prizesCount))
   392:       ? Number(d.prizesCount)
   393:       : 0;
   394:     const hasLogical = !!(d?.ymd && d?.close_hour);
   395:     return prizesLen * 1_000_000 + pc * 1_000 + (hasLogical ? 10 : 0);
   396:   }
   397: 
   398:   for (let i = 0; i < arr.length; i += 1) {
   399:     const raw = arr[i] || {};
   400: 
   401:     const ymd = raw.ymd || normalizeToYMD(raw.date) || "";
   402:     const hour = normalizeHourLike(raw.close_hour || raw.closeHour || "");
   403: 
   404:     const lotCode = normalizeLotteryCodeAny(raw);
   405: 
   406:     const drawId = raw.drawId ?? raw.id ?? raw.__name__ ?? null;
   407:     const idStr = drawId != null ? String(drawId) : "";
   408: 
   409:     const hasLogical = !!(ymd && hour);
   410: 
   411:     const key = hasLogical
   412:       ? lotCode
   413:         ? `${ymd}__${hour}__${lotCode}`
   414:         : `${ymd}__${hour}`
   415:       : `id__${idStr || `idx_${i}`}`;
   416: 
   417:     const normalized = {
   418:       ...raw,
   419:       ymd: ymd || raw.ymd || null,
   420:       close_hour: hour || raw.close_hour || raw.closeHour || "",
   421:       closeHour: hour || raw.closeHour || raw.close_hour || "",
   422:       lottery_code: lotCode || raw.lottery_code || raw.lotteryCode || null,
   423:       drawId: raw.drawId || raw.id || raw.__name__ || null,
   424:       id: raw.id || raw.drawId || raw.__name__ || null,
   425:     };
   426: 
   427:     if (!byKey.has(key)) {
   428:       byKey.set(key, normalized);
   429:       order.push(key);
   430:       continue;
   431:     }
   432: 
   433:     const prev = byKey.get(key);
   434:     if (score(normalized) > score(prev)) {
   435:       byKey.set(key, normalized);
   436:     }
   437:   }
   438: 
   439:   return order.map((k) => byKey.get(k)).filter(Boolean);

   443:   const arr = Array.isArray(items) ? items : [];
   444:   const concurrency = Math.max(1, Number(limitN) || 6);
   445:   const results = new Array(arr.length);

   447:
