  383: function normalizeCloseHourForLottery(value, lotteryKey) {
  384:   const lk = String(lotteryKey || "").trim().toUpperCase();
  385:   const raw0 = normalizeHHMM(value);
  386:   if (!raw0 || !isHHMM(raw0)) return { raw: "", slot: "" };
  387: 
  388:   // ✅ FEDERAL: slot de negócio é sempre 20:00, mas preserva o raw
  389:   if (lk === "FEDERAL") {
  390:     return { raw: raw0, slot: "20:00" };
  391:   }
  392: 
  393:   if (lk === "PT_RIO") {
  394:     const hh = raw0.slice(0, 2);
  395:     const mm = raw0.slice(3, 5);
  396: 
  397:     // ✅ se vier HH:09, é marcação (não minuto real) -> não grava close_hour_raw
  398:     const raw = mm === "09" ? "" : raw0;
  399:     return { raw, slot: `${hh}:00` };
  400:   }
  401: 
  402:   return { raw: raw0, slot: raw0 };
  403: }
