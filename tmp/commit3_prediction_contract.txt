   28:   "25": "Vaca",
   29: };
   30: 
   31: function pad2(n) {
   32:   return String(n).padStart(2, "0");
   33: }
   34: 
   35: function toNum(v) {
   36:   const n = Number(v);
   37:   return Number.isFinite(n) ? n : 0;
   38: }
   39: 
   40: function rankingToPredictions(ranking = [], opts = {}) {
   41:   const limit = Math.max(1, Number(opts.limit || 10));
   42: 
   43:   const rows = (Array.isArray(ranking) ? ranking : [])
   44:     .map((item) => {
   45:       const grupo = pad2(item.grupo || item.grupoNum || item.group || item.id);
   46:       const aparicoes = toNum(item.total ?? item.apar ?? item.count ?? 0);
   47: 
   48:       return {
   49:         grupo,
   50:         animal: item.animal || item.nome || ANIMALS[grupo] || null,
   51:         aparicoes,
   52:         raw: item,
   53:       };
   54:     })
   55:     .filter((x) => /^\d{2}$/.test(x.grupo))
   56:     .sort((a, b) => b.aparicoes - a.aparicoes || Number(a.grupo) - Number(b.grupo))
   57:     .slice(0, limit);
   58: 
   59:   const max = rows.reduce((m, x) => Math.max(m, x.aparicoes), 0) || 1;
   60: 
   61:   return rows.map((x, idx) => {
   62:     const score = Math.round((x.aparicoes / max) * 100);
   63:     const confidence = Math.max(1, Math.round(score - idx * 2));
   64: 
   65:     return {
   66:       type: "grupo",
   67:       grupo: x.grupo,
   68:       animal: x.animal,
   69:       score,
   70:       confidence,
   71:       reasons: [
   72:         `Ranking V1: ${x.aparicoes} aparições no recorte analisado`,
   73:         `Posição ${idx + 1} entre os grupos avaliados`,
   74:       ],
   75:       signals: {
   76:         aparicoes: x.aparicoes,
   77:         rankPosition: idx + 1,
   78:         engine: "score_engine_v1",
   79:       },
   80:     };
   81:   });
   82: }
   83: 
   84: module.exports = {
   85:   rankingToPredictions,
   86: };
