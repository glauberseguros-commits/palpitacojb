   39: function groupDrawsByMonth(draws = []) {
   40:   const grouped = new Map();
   41: 
   42:   for (const rawDraw of safeArray(draws)) {
   43:     const draw = normalizeDraw(rawDraw);
   44: 
   45:     if (!draw) {
   46:       continue;
   47:     }

   49:     const yearMonth = draw.ymd.slice(0, 7);
   50: 
   51:     if (!grouped.has(yearMonth)) {
   52:       grouped.set(yearMonth, []);
   53:     }

   55:     grouped.get(yearMonth).push(draw);
   56:   }

   58:   const normalized = new Map();
