   442: async function mapWithConcurrency(items, limitN, mapper) {
   443:   const arr = Array.isArray(items) ? items : [];
   444:   const concurrency = Math.max(1, Number(limitN) || 6);
   445:   const results = new Array(arr.length);
   446:   let idx = 0;
   447: 
   448:   async function worker() {
   449:     while (true) {
   450:       const current = idx;
   451:       idx += 1;
   452:       if (current >= arr.length) break;
   453:       results[current] = await mapper(arr[current], current);
   454:     }
   455:   }
   456: 
   457:   await Promise.all(
   458:     Array.from({ length: Math.min(concurrency, arr.length) }, () => worker())
