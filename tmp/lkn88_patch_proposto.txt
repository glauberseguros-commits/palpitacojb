function normalizeDrawCloseHour(draw, lotteryKey) {

  const base = normalizeCloseHourForLottery(
      draw?.close_hour || "",
      lotteryKey
  );

  const lk = String(lotteryKey || "").trim().toUpperCase();

  if (lk !== "NACIONAL") {
      return base;
  }

  const lotteryName = String(
      draw?.lottery_name ||
      draw?.name ||
      ""
  );

  const match = lotteryName.match(/(\d{2})\s*HS/i);

  if (!match) {
      return base;
  }

  return {
      raw: base.raw,
      slot: `${match[1]}:00`
  };
}
