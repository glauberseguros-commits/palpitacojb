   44: function dayDiffInclusiveUTC(ymdFrom, ymdTo) {
   45:   const toUTC = (ymd) => {
   46:     const m = String(ymd).match(/^(\d{4})-(\d{2})-(\d{2})$/);
   47:     if (!m) return null;
   48:     return new Date(Date.UTC(Number(m[1]), Number(m[2]) - 1, Number(m[3])));
   49:   };
   50: 
   51:   const da = toUTC(ymdFrom);
   52:   const db = toUTC(ymdTo);
   53:   if (!da || !db) return NaN;
   54: 
   55:   const ms = db.getTime() - da.getTime();
   56:   if (!Number.isFinite(ms)) return NaN;
   57: 
   58:   return Math.floor(ms / 86400000) + 1;
   59: }

   74: function normHHMM(value) {
   75:   const s = String(value || "").trim();
   76:   if (!s) return "";
   77: 
   78:   const mh = s.match(/^(\d{1,2})h$/i);
   79:   if (mh) return `${String(mh[1]).padStart(2, "0")}:00`;
   80: 
   81:   const mISO = s.match(/^(\d{1,2}):(\d{1,2})(?::(\d{1,2}))?$/);
   82:   if (mISO) {
   83:     const hh = String(mISO[1]).padStart(2, "0");
   84:     const mm = String(mISO[2]).padStart(2, "0");
   85:     return `${hh}:${mm}`;
   86:   }
   87: 
   88:   const m2 = s.match(/^(\d{1,2})$/);
   89:   if (m2) return `${String(m2[1]).padStart(2, "0")}:00`;
   90: 
   91:   return s;
   92: }

  130: function dedupeDrawsLogicalPreferBest(draws) {
  131:   const arr = Array.isArray(draws) ? draws : [];
  132:   if (!arr.length) return arr;
  133: 
  134:   const byKey = new Map();
  135:   const order = [];
  136: 
  137:   const score = (d) => {
  138:     const prizesLen = Array.isArray(d?.prizes) ? d.prizes.length : 0;
  139:     const pc = Number.isFinite(Number(d?.prizesCount)) ? Number(d.prizesCount) : 0;
  140:     const ymd = normalizeToYMD(getDrawDateRaw(d));
  141:     const hhmm = normHHMM(getDrawHourRaw(d));
  142:     const hasLogical = !!(ymd && hhmm);
  143:     return prizesLen * 1000000 + pc * 1000 + (hasLogical ? 10 : 0);
  144:   };
  145: 
  146:   for (let i = 0; i < arr.length; i += 1) {
  147:     const d = arr[i] || {};
  148:     const ymd = normalizeToYMD(getDrawDateRaw(d)) || "";
  149:     const hhmm = normHHMM(getDrawHourRaw(d)) || "";
  150:     const lotteryCode = getDrawLotteryCodeRaw(d) || "";
  151: 
  152:     const id =
  153:       (d?.drawId != null && String(d.drawId)) ||
  154:       (d?.id != null && String(d.id)) ||
  155:       "";
  156: 
  157:     const hasLogical = !!(ymd && hhmm);
  158: 
  159:     const key = hasLogical
  160:       ? lotteryCode
  161:         ? `${ymd}__${hhmm}__${lotteryCode}`
  162:         : `${ymd}__${hhmm}`
  163:       : `id__${id || `idx_${i}`}`;
  164: 
  165:     if (!byKey.has(key)) {
  166:       byKey.set(key, d);
  167:       order.push(key);
  168:       continue;
  169:     }
  170: 
  171:     const prev = byKey.get(key);
  172:     if (score(d) > score(prev)) {
  173:       byKey.set(key, d);
  174:     }
  175:   }
  176: 
  177:   return order.map((k) => byKey.get(k)).filter(Boolean);
  178: }

  202: function clampYmdToBounds(ymd, minYmd, maxYmd) {
  203:   if (!ymd) return null;
  204: 
  205:   const n = ymdToNumber(ymd);
  206:   const a = ymdToNumber(minYmd);
  207:   const b = ymdToNumber(maxYmd);

  210: 
  211:   if (Number.isFinite(a) && n < a) return minYmd;
  212:   if (Number.isFinite(b) && n > b) return maxYmd;
  213:   return ymd;

  216: function clampRangeToBounds(ymdFrom, ymdTo, minYmd, maxYmd) {
  217:   let a = ymdFrom || null;
  218:   let b = ymdTo || null;
  219: 
  220:   if (minYmd || maxYmd) {
  221:     a = clampYmdToBounds(a, minYmd, maxYmd);
  222:     b = clampYmdToBounds(b, minYmd, maxYmd);
  223:   }

  225:   const na = ymdToNumber(a);
  226:   const nb = ymdToNumber(b);
  227:   if (Number.isFinite(na) && Number.isFinite(nb) && na > nb) {
  228:     return { ymdFrom: b, ymdTo: a };
  229:   }
  230: 
  231:   return { ymdFrom: a, ymdTo: b };
