02954:   const out = new Map();

02967: function layerProbability(freqMap, samples, alpha = TOP3_SMOOTH_ALPHA) {
02968:   const out = new Map();
02969:   const k = safeInt(TOP3_GROUPS_K, 25);
02970:   const a = Math.max(0, Number(alpha || 0));
02971:   const base = Math.max(0, Number(samples || 0));
02972:   const denom = Math.max(1, base + (a * k));
02973: 
02974:   for (let g = 1; g <= k; g += 1) {
02975:     const count = Math.max(0, Number(freqMap?.get?.(g) || 0));
02976:     out.set(g, (count + a) / denom);
02977:   }
02978: 
02979:   return out;
02980: }

02984:   const lim = Math.max(1, Number(fullAt || 30));
02985:   return Math.max(0, Math.min(1, n / lim));

04341: function buildTimelineForDate({
04342:   ymd,
04343:   drawsToday,
04344:   drawsRange,
04345:   lotteryKey,
04346:   PT_RIO_SCHEDULE_NORMAL,
04347:   PT_RIO_SCHEDULE_WED_SAT,
04348:   FEDERAL_SCHEDULE,
04349: }) {
04350:   const targetYmd = safeStr(ymd);
04351:   const day = Array.isArray(drawsToday) ? [...drawsToday] : [];
04352:   const range = Array.isArray(drawsRange) ? [...drawsRange] : [];
04353: 
04354:   if (!isYMD(targetYmd) || !range.length) return [];
04355: 
04356:   const schedule = getScheduleForLottery({
04357:     lotteryKey,
04358:     ymd: targetYmd,
04359:     PT_RIO_SCHEDULE_NORMAL,
04360:     PT_RIO_SCHEDULE_WED_SAT,
04361:     FEDERAL_SCHEDULE,
04362:   });
04363: 
04364:   const normalizedSchedule = (Array.isArray(schedule) ? schedule : [])
04365:     .map(toHourBucket)
04366:     .filter(Boolean);
04367: 
04368:   if (!normalizedSchedule.length) return [];
04369: 
04370:   const daySorted = day
04371:     .filter((d) => isYMD(pickDrawYMD(d)) && toHourBucket(pickDrawHour(d)))
04372:     .sort((a, b) => {
04373:       const ta = ymdHourToTs(pickDrawYMD(a), toHourBucket(pickDrawHour(a)));
04374:       const tb = ymdHourToTs(pickDrawYMD(b), toHourBucket(pickDrawHour(b)));
04375:       return ta - tb;
04376:     });
04377: 
04378:   const rangeSorted = range
04379:     .filter((d) => isYMD(pickDrawYMD(d)) && toHourBucket(pickDrawHour(d)))
04380:     .sort((a, b) => {
04381:       const ta = ymdHourToTs(pickDrawYMD(a), toHourBucket(pickDrawHour(a)));
04382:       const tb = ymdHourToTs(pickDrawYMD(b), toHourBucket(pickDrawHour(b)));
04383:       return ta - tb;
04384:     });
04385: 
04386:   const realizedTodayMap = new Map();
04387: 
04388:   for (const d of daySorted) {
04389:     const y = pickDrawYMD(d);
04390:     const h = toHourBucket(pickDrawHour(d));
04391:     if (!isYMD(y) || !h) continue;
04392:     if (y !== targetYmd) continue;
04393:     realizedTodayMap.set(`${y}|${h}`, d);
04394:   }
04395: 
04396: 
04397:   const chainContextDraws = [];
04398:   const timeline = [];
04399:   let pendingPredictionAlreadyShown = false;
04400: 
04401:   for (const slotHour of normalizedSchedule) {
04402:     const currentDraw = realizedTodayMap.get(`${targetYmd}|${slotHour}`) || null;
04403:     const slotTs = ymdHourToTs(targetYmd, slotHour);
04404: 
04405: 
04406:     if (!currentDraw && pendingPredictionAlreadyShown) {
04407:       timeline.push({
04408:         targetYmd,
04409:         targetHour: slotHour,
04410:         baseYmd: "",
04411:         baseHour: "",
04412:         top3: [],
04413:         resultGrupo: null,
04414:         hit: null,
04415:         status: "pending",
04416:       });
04417:       continue;
04418:     }
04419: 
04420:     let baseDraw = null;
04421: 
04422:     const priorChainDraw = [...chainContextDraws]
04423:       .filter((d) => {
04424:         const y = pickDrawYMD(d);
04425:         const h = toHourBucket(pickDrawHour(d));
04426:         const ts = ymdHourToTs(y, h);
04427:         return isYMD(y) && h && Number.isFinite(ts) && ts < slotTs;
04428:       })
04429:       .sort((a, b) => {
04430:         const ta = ymdHourToTs(pickDrawYMD(a), toHourBucket(pickDrawHour(a)));
04431:         const tb = ymdHourToTs(pickDrawYMD(b), toHourBucket(pickDrawHour(b)));
04432:         return tb - ta;
04433:       })[0] || null;
04434: 
04435:     if (priorChainDraw) baseDraw = priorChainDraw;
04436: 
04437:     if (!baseDraw) {
04438:       const prevSameDay = daySorted
04439:         .filter((d) => {
04440:           const y = pickDrawYMD(d);
04441:           const h = toHourBucket(pickDrawHour(d));
04442:           const ts = ymdHourToTs(y, h);
04443:           return y === targetYmd && h && Number.isFinite(ts) && ts < slotTs;
04444:         })
04445:         .sort((a, b) => {
04446:           const ta = ymdHourToTs(pickDrawYMD(a), toHourBucket(pickDrawHour(a)));
04447:           const tb = ymdHourToTs(pickDrawYMD(b), toHourBucket(pickDrawHour(b)));
04448:           return tb - ta;
04449:         })[0] || null;
04450: 
04451:       if (prevSameDay) baseDraw = prevSameDay;
04452:     }
04453: 
04454:     if (!baseDraw) {
04455:       for (let i = rangeSorted.length - 1; i >= 0; i -= 1) {
04456:         const d = rangeSorted[i];
04457:         const y = pickDrawYMD(d);
04458:         const h = toHourBucket(pickDrawHour(d));
04459:         const ts = ymdHourToTs(y, h);
04460: 
04461:         if (!Number.isFinite(ts) || ts >= slotTs) continue;
04462: 
04463:         baseDraw = d;
04464:         break;
04465:       }
04466:     }
04467: 
04468:     if (!baseDraw) {
04469:       baseDraw = findPreviousValidDraw(rangeSorted, targetYmd, slotHour);
04470:     }
04471: 
04472:     if (!baseDraw) {
04473:       timeline.push({
04474:         targetYmd,
04475:         targetHour: slotHour,
04476:         baseYmd: "",
04477:         baseHour: "",
04478:         top3: [],
04479:         resultGrupo: null,
04480:         hit: null,
04481:         status: "pending",
04482:       });
04483:       continue;
04484:     }
04485: 
04486:     const baseTs = ymdHourToTs(
04487:       pickDrawYMD(baseDraw),
04488:       toHourBucket(pickDrawHour(baseDraw))
04489:     );
04490: 
04491:     const usableHistory = rangeSorted.filter((d) => {
04492:       const ts = ymdHourToTs(
04493:         pickDrawYMD(d),
04494:         toHourBucket(pickDrawHour(d))
04495:       );
04496:       return Number.isFinite(ts) && ts <= baseTs;
04497:     });
04498: 
04499:     const historyStats = (() => {
04500:       const MS_DAY = 24 * 60 * 60 * 1000;
04501: 
04502:       let d30 = 0;
04503:       let d90 = 0;
04504:       let d180 = 0;
04505:       let d365 = 0;
04506: 
04507:       for (const historicalDraw of usableHistory) {
04508:         const historicalTs = ymdHourToTs(
04509:           pickDrawYMD(historicalDraw),
04510:           toHourBucket(pickDrawHour(historicalDraw))
04511:         );
04512: 
04513:         if (!Number.isFinite(historicalTs)) continue;
04514: 
04515:         const diffDays = (baseTs - historicalTs) / MS_DAY;
04516: 
04517:         if (diffDays >= 0 && diffDays <= 30) d30 += 1;
04518:         if (diffDays >= 0 && diffDays <= 90) d90 += 1;
04519:         if (diffDays >= 0 && diffDays <= 180) d180 += 1;
04520:         if (diffDays >= 0 && diffDays <= 365) d365 += 1;
04521:       }
04522: 
04523:       return {
04524:         total: usableHistory.length,
04525:         d30,
04526:         d90,
04527:         d180,
04528:         d365,
04529:       };
04530:     })();
04531: 
04532:     const usableTodayContext = [...chainContextDraws]
04533:       .filter((d) => {
04534:         const ts = ymdHourToTs(
04535:           pickDrawYMD(d),
04536:           toHourBucket(pickDrawHour(d))
04537:         );
04538:         return Number.isFinite(ts) && ts <= baseTs;
04539:       })
04540:       .sort((a, b) => {
04541:         const ta = ymdHourToTs(pickDrawYMD(a), toHourBucket(pickDrawHour(a)));
04542:         const tb = ymdHourToTs(pickDrawYMD(b), toHourBucket(pickDrawHour(b)));
04543:         return ta - tb;
04544:       });
04545: 
04546:     const computed = computeStatisticalTop3V3({
04547:       lotteryKey,
04548:       drawsRange: [...usableHistory, ...usableTodayContext],
04549:       drawLast: baseDraw,
04550:       drawsToday: usableTodayContext,
04551:       PT_RIO_SCHEDULE_NORMAL,
04552:       PT_RIO_SCHEDULE_WED_SAT,
04553:       FEDERAL_SCHEDULE,
04554:       topN: 3,
04555:       targetYmdOverride: targetYmd,
04556:       targetHourOverride: slotHour,
04557:     });
04558: 
04559: 
04560:     const computedNextYmd = safeStr(computed?.meta?.next?.ymd || "");
04561:     const computedNextHour = toHourBucket(computed?.meta?.next?.hour || "");
04562:     const computedForSlot =
04563:       computedNextYmd === targetYmd && computedNextHour === slotHour;
04564:

04566:       computedForSlot && Array.isArray(computed?.top)
04567:         ? computed.top.slice(0, 3)
04568:         : [];
04569: 
04570:     const resultTop3Groups = currentDraw
04571:       ? getPrizeGroupsByPosition(currentDraw, 3)

04574:     const resultGrupo = Number(resultTop3Groups[0]);

04576:     const normalizedResultGrupo =
04577:       Number.isFinite(resultGrupo) &&
04578:       resultGrupo >= 1 &&
04579:       resultGrupo <= TOP3_GROUPS_K
04580:         ? resultGrupo
04581:         : null;
04582: 
04583:     const hitPrizePositions = resultTop3Groups.map(
04584:       (resultGroup) =>
04585:         Number.isFinite(Number(resultGroup)) &&
04586:         top3.some(
04587:           (item) =>
04588:             Number(item?.grupo) ===
04589:             Number(resultGroup)
04590:         )
04591:     );
04592: 
04593:     const matchedPrizePositions =
04594:       hitPrizePositions.filter(Boolean).length;
04595: 
04596:     const hit =
04597:       top3.length > 0 &&
04598:       resultTop3Groups.some(
04599:         (resultGroup) =>
04600:           Number.isFinite(Number(resultGroup))
04601:       )
04602:         ? matchedPrizePositions > 0
04603:         : null;
04604: 
04605:     timeline.push({
04606:       targetYmd,
04607:       targetHour: slotHour,
04608: 
04609:       baseYmd: pickDrawYMD(baseDraw) || "",
04610:       baseHour: toHourBucket(pickDrawHour(baseDraw)) || "",
04611: 
04612:       top3,
04613: 
04614:       resultGrupo: normalizedResultGrupo,
04615:       resultTop3Groups,
04616: 
04617:       hit,
04618:       top3PrizeHit: hit,
04619:       hitPrizePositions,
04620:       matchedPrizePositions,
04621: 
04622:       status: Number.isFinite(
04623:         Number(normalizedResultGrupo)
04624:       )
04625:         ? "validated"

04627: 
04628:       historyStats,
04629:     });
04630: 
04631:     if (currentDraw) {
04632:       chainContextDraws.push(currentDraw);
04633:     } else {

04635:     }
04636:   }
