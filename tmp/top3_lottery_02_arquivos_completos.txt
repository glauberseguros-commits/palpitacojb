02947: function incMap(map, key, value = 1) {
02948:   const k = Number(key);
02949:   if (!Number.isFinite(k) || k < 1 || k > TOP3_GROUPS_K) return;
02950:   map.set(k, Number(map.get(k) || 0) + Number(value || 0));
02951: }
02952: 
02953: function emptyGroupMap() {
02954:   const out = new Map();
02955:   for (let g = 1; g <= safeInt(TOP3_GROUPS_K, 25); g += 1) out.set(g, 0);
02956:   return out;
02957: }
02958: 
02959: function getPrizePresenceGroupSet(draw) {
02960:   return new Set(
02961:     getAllPrizePresenceGroups(draw)
02962:       .map((x) => Number(x.grupo))
02963:       .filter((g) => Number.isFinite(g) && g >= 1 && g <= TOP3_GROUPS_K)
02964:   );
02965: }
02966: 
02967: function layerProbability(freqMap, samples, alpha = TOP3_SMOOTH_ALPHA) {
02968:   const out = new Map();
02969:   const k = safeInt(TOP3_GROUPS_K, 25);
02970:   const a = Math.max(0, Number(alpha || 0));
02971:   const base = Math.max(0, Number(samples || 0));
02972:   const denom = Math.max(1, base + (a * k));
02973: 
02974:   for (let g = 1; g <= k; g += 1) {
02975:     const count = Math.max(0, Number(freqMap?.get?.(g) || 0));
02976:     out.set(g, (count + a) / denom);
02977:   }
02978: 
02979:   return out;
02980: }
02981: 
02982: function sampleConfidence(samples, fullAt = 30) {
02983:   const n = Number(samples || 0);
02984:   const lim = Math.max(1, Number(fullAt || 30));
02985:   return Math.max(0, Math.min(1, n / lim));
02986: }
