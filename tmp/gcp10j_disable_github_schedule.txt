name: Auto Import ÔÇö PalPitaco JB

on:
  # Automatic scheduling migrated to Google Cloud Scheduler.

  workflow_dispatch:
    inputs:
      lottery:
        description: "Qual loteria executar"
        required: true
        default: "ALL"
        type: choice
        options:
          - ALL
          - PT_RIO
          - FEDERAL
          - LOOK
          - NACIONAL

# Mant├®m somente uma execu├º├úo ativa e uma pendente.
# Cada execu├º├úo reconcilia todas as loterias e recupera slots anteriores.
concurrency:
  group: auto-import-heartbeat-${{ github.ref }}
  cancel-in-progress: false

permissions:
  contents: read

env:
  TZ: America/Sao_Paulo
  PITACO_API_BASE: ${{ vars.PITACO_API_BASE }}

jobs:
  reconcile-results:
    name: Reconciliar resultados
    runs-on: ubuntu-latest
    timeout-minutes: 30

    defaults:
      run:
        working-directory: backend

    steps:
      - name: Checkout do reposit├│rio
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm
          cache-dependency-path: backend/package-lock.json

      - name: Instalar depend├¬ncias
        run: npm ci

      - name: Importar hoje e reconciliar ontem
        shell: bash
        env:
          FIREBASE_ADMIN_SA_JSON: ${{ secrets.FIREBASE_SERVICE_ACCOUNT_JSON }}
          FAIL_ON_CRITICAL: "1"
          AUDIT_WARN_MIN: "20"
          AUDIT_CRIT_MIN: "60"
          CATCHUP_MAX_AFTER_END_MIN: "1440"
        run: |
          set +e

          requested="${{ inputs.lottery }}"

          if [[ "${{ github.event_name }}" == "workflow_dispatch" && "$requested" != "ALL" ]]; then
            lotteries="$requested"
          else
            lotteries="PT_RIO FEDERAL LOOK NACIONAL"
          fi

          yesterday="$(TZ=America/Sao_Paulo date -d 'yesterday' +%F)"
          overall=0

          run_import() {
            local lottery="$1"
            local period="$2"
            shift 2

            echo ""
            echo "=================================================="
            echo "LOTTERY=$lottery | PERIOD=$period"
            echo "=================================================="

            env LOTTERY="$lottery" "$@" node scripts/autoImportToday.js

            if [[ $code -ne 0 ]]; then
              echo "::error::Falha LOTTERY=$lottery PERIOD=$period EXIT=$code"

            fi

          for lottery in $lotteries; do
            run_import "$lottery" "TODAY"
          done

          for lottery in $lotteries; do
            run_import \
              "$lottery" \
              "YESTERDAY:$yesterday" \
              DATE="$yesterday" \
              NOW_HM="23:59"
          done

          exit $overall
