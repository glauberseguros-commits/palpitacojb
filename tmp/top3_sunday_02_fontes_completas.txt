02859: function layerProbability(freqMap, samples, alpha = TOP3_SMOOTH_ALPHA) {
02860:   const out = new Map();
02861:   const k = safeInt(TOP3_GROUPS_K, 25);
02862:   const a = Math.max(0, Number(alpha || 0));
02863:   const base = Math.max(0, Number(samples || 0));
02864:   const denom = Math.max(1, base + (a * k));
02865: 
02866:   for (let g = 1; g <= k; g += 1) {
02867:     const count = Math.max(0, Number(freqMap?.get?.(g) || 0));
02868:     out.set(g, (count + a) / denom);
02869:   }
02870: 
02871:   return out;

03327: function compareScenes(a, b, explain = false) {
03328:   if (!a || !b) return explain ? { score: 0, details: {} } : 0;
03329: 
03330:   let score = 0;
03331:   const details = {};
03332: 
03333:   const add = (key, value) => {
03334:     if (!value) return;
03335:     score += value;
03336:     details[key] = value;
03337:   };
03338: 
03339:   const ag = Array.isArray(a.grupos) ? a.grupos : [];
03340:   const bg = Array.isArray(b.grupos) ? b.grupos : [];

03343:   const bd = Array.isArray(b.dezenas) ? b.dezenas : [];

03346:   const bc = Array.isArray(b.centenas) ? b.centenas : [];
03347: 
03348:   if (Number(a.firstGrupo) === Number(b.firstGrupo))
03349:     add("firstGrupo",45);
03350: 
03351:   for (let i = 0; i < Math.min(ag.length,bg.length,7); i++) {
03352:     if (Number(ag[i]) === Number(bg[i]))

03356:   add(
03357:     "groupOverlap",
03358:     new Set(ag.filter(g => bg.includes(g))).size * 6

03381:     add("signature",60);
03382: 
03383:   if (a.hour === b.hour)

03386:   return explain ? { score, details } : score;

00998:   }));
00999: 
01000:   for (const d of list) {
01001:     const prevY = pickDrawYMD(d);
01002:     const prevH = toHourBucket(pickDrawHour(d));
01003:     if (!isYMD(prevY) || !prevH) continue;
01004:

01019:     const found = findNextExistingDrawFromSlot({
01020:       lotteryKey,
01021:       startSlot: { ymd: ns.ymd, hour: ns.hour },
01022:       drawsIndex,

01024:       PT_RIO_SCHEDULE_WED_SAT,
01025:       FEDERAL_SCHEDULE,
01026:     });
01027:

01029:     const nextSlot = found?.slot || null;
01030:     if (!nextDraw || !nextSlot?.ymd || !nextSlot?.hour) continue;

01033:       prevYmd: prevY,
01034:       prevHour: prevH,

01056:     layerResults.find((x) => x.samples >= x.minSamples && x.freq.size > 0) ||

01062:       key: "NONE",
01063:       label: "sem amostra",
01064:       minSamples: 0,
01065:       samples: 0,
01066:       freq: new Map(),
01067:       firstFreq: new Map(),
01068:       matchingDraws: [],
01069:     };
01070:   }
01071: 
01072:   return {
01073:     chosen,
01074:     layers: layerResults,
01075:   };
