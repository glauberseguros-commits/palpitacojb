   512: async function mapWithConcurrency(items, limitN, mapper) {
   513:   const arr = Array.isArray(items) ? items : [];
   514:   const concurrency = Math.max(1, Number(limitN) || 6);
   515:   const results = new Array(arr.length);
   516:   let idx = 0;
   517: 
   518:   async function worker() {
   519:     while (true) {
   520:       const current = idx;
   521:       idx += 1;
   522:       if (current >= arr.length) break;
   523:       results[current] = await mapper(arr[current], current);
   524:     }
   525:   }
   526: 
   527:   await Promise.all(
   528:     Array.from({ length: Math.min(concurrency, arr.length) }, () => worker())
   529:   );
   530:   return results;
   531: }
